#include "library.h"
#include "sut_common.h"

#include <string>
#include <sstream>

extern "C" {

char* format_list(const int* input, size_t input_length) {
    if (input == nullptr && input_length != 0) {
        return nullptr;
    }

    try {
        std::string result;
        result.reserve(input_length * 12 + 2); // Reserve space for numbers and commas
        result += '[';
        for (size_t i = 0; i < input_length; ++i) {
            if (i > 0) result += ", ";
            result += std::to_string(input[i]);
        }
        result += ']';

        return copy_to_c_string(result);
    } catch (...) {
        return nullptr;
    }
}

}
