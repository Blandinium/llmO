#include "library.h"

extern "C" {

uint64_t fibonacci(uint64_t n) {
    if (n <= 1) {
        return n;
    }
    
    uint64_t a = 0;
    uint64_t b = 1;
    uint64_t c;
    
    // Use a more optimized loop with explicit unrolling
    for (uint64_t i = 2; i <= n; i += 2) {
        c = a + b;
        a = b;
        b = c;
        
        if (i + 1 <= n) {
            c = a + b;
            a = b;
            b = c;
        }
    }
    
    return b;
}

}
