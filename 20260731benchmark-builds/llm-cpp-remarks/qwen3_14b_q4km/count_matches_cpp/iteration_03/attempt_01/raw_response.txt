#include "library.h"
#include <algorithm>
#include <vector>

extern "C" {

size_t count_matches(
    const int* allowed,
    size_t allowed_length,
    const int* queries,
    size_t queries_length
) {
    if ((allowed == nullptr && allowed_length != 0) ||
        (queries == nullptr && queries_length != 0)) {
        return 0;
    }

    // Use a vector for better cache locality and to avoid hash table overhead
    std::vector<int> allowed_vector;
    if (allowed_length > 0) {
        allowed_vector.assign(allowed, allowed + allowed_length);
    }

    // Sort the allowed values for binary search
    std::sort(allowed_vector.begin(), allowed_vector.end());

    size_t matches = 0;
    for (size_t i = 0; i < queries_length; ++i) {
        // Use binary search to check if query is in allowed
        if (std::binary_search(allowed_vector.begin(), allowed_vector.end(), queries[i])) {
            ++matches;
        }
    }

    return matches;
}

}
