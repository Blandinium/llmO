#include "library.h"

extern "C" {

uint64_t fibonacci(uint64_t n) {
    if (n <= 1) {
        return n;
    }
    
    uint64_t a = 0;
    uint64_t b = 1;
    uint64_t i = 2;
    
    // Unroll the loop manually to help vectorization
    while (i <= n) {
        uint64_t next = a + b;
        a = b;
        b = next;
        ++i;
        
        if (i <= n) {
            uint64_t next2 = a + b;
            a = b;
            b = next2;
            ++i;
        }
    }
    
    return b;
}

}
